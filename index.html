/* --- 1. GLASSMORPHISM FROSTED GLASS EFFECT --- */
.glass-card {
  background: rgba(23, 31, 51, 0.45);
  backdrop-filter: blur(14px);
  -webkit-backdrop-filter: blur(14px);
  border: 1px solid rgba(255, 255, 255, 0.12);
  box-shadow: 0 8px 32px 0 rgba(0, 0, 0, 0.37);
  border-radius: 16px;
}
        
/* --- 2. DARK HYBRID NEUMORPHISM (RAISED & PRESSED) --- */
.neu-raised {
  background: linear-gradient(145deg, #121a2d, #0e1423);
  box-shadow: 10px 10px 20px #05080e, 
              -10px -10px 20px #17223b;
  border-radius: 16px;
}

.neu-pressed {
  background: #0d1322;
  box-shadow: inset 8px 8px 12px #060910, 
              inset -8px -8px 12px #141d34;
  border-radius: 16px;
}
        
/* --- 3. VIVID GRADIENTS & HOVER BOX GLOW --- */
.gradient-text {
  background: linear-gradient(135deg, #a855f7, #ec4899, #3b82f6);
  -webkit-background-clip: text;
  -webkit-text-fill-color: transparent;
}

.glow-hover:hover {
  box-shadow: 0 0 20px rgba(168, 85, 247, 0.6),
              0 0 40px rgba(168, 85, 247, 0.3);
  transition: all 0.3s ease;
}
        
/* --- 4. CSS KEYFRAMES & ORBIT ANIMATIONS --- */
@keyframes orbit {
  from { transform: rotate(0deg) translateX(50px) rotate(0deg); }
  to   { transform: rotate(360deg) translateX(50px) rotate(-360deg); }
}

.animated-orbit {
  animation: orbit 6s linear infinite;
}
